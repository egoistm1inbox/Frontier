cmake_minimum_required(VERSION 3.24)
# Standalone diagnostic only; does not replace Slate's MSVC engine build route.
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreadedDLL")
project(NetworkingLoginWindow LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
foreach(REQUIRED_ROOT EOS_SDK_ROOT GUI_ROOT PHOTON_ROOT)
    if(NOT DEFINED ${REQUIRED_ROOT})
        message(FATAL_ERROR "Supply ${REQUIRED_ROOT}")
    endif()
endforeach()
file(TO_CMAKE_PATH "${EOS_SDK_ROOT}" EOS_SDK_ROOT)
file(TO_CMAKE_PATH "${GUI_ROOT}" GUI_ROOT)
file(TO_CMAKE_PATH "${PHOTON_ROOT}" PHOTON_ROOT)
set(PHOTON_INCLUDE "${PHOTON_ROOT}/realtime/include")
if(NOT EXISTS "${PHOTON_INCLUDE}/LoadBalancing-cpp/inc/Client.h")
    message(FATAL_ERROR "Photon headers missing under ${PHOTON_INCLUDE}")
endif()
if(NOT DEFINED PHOTON_LIB_DIR)
    set(PHOTON_LIB_DIR "${EOS_SDK_ROOT}/Photon/Lib")
endif()
file(TO_CMAKE_PATH "${PHOTON_LIB_DIR}" PHOTON_LIB_DIR)
file(GLOB PHOTON_LIBS "${PHOTON_LIB_DIR}/*.lib")
list(LENGTH PHOTON_LIBS PHOTON_LIB_COUNT)
if(PHOTON_LIB_COUNT GREATER 0 AND PHOTON_LIB_COUNT LESS 3)
    message(FATAL_ERROR "Partial Photon lib set in ${PHOTON_LIB_DIR} (need Common, Photon and LoadBalancing /MD x64, or none for stub mode).")
endif()
if(PHOTON_LIB_COUNT GREATER_EQUAL 3)
    set(PHOTON_LINKED TRUE)
    message(STATUS "Photon: linking ${PHOTON_LIBS}")
else()
    message(STATUS "Photon: official /MD x64 libs absent; linking the stub (compile check still runs).")
endif()
set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GLFW_INSTALL OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory("${GUI_ROOT}/glfw" glfw)
find_package(OpenGL REQUIRED)
set(IMGUI "${GUI_ROOT}/imgui")
add_library(LoginImGui STATIC
    "${IMGUI}/imgui.cpp" "${IMGUI}/imgui_draw.cpp"
    "${IMGUI}/imgui_tables.cpp" "${IMGUI}/imgui_widgets.cpp"
    "${IMGUI}/backends/imgui_impl_glfw.cpp" "${IMGUI}/backends/imgui_impl_opengl2.cpp")
target_include_directories(LoginImGui PUBLIC "${IMGUI}" "${IMGUI}/backends")
target_link_libraries(LoginImGui PUBLIC glfw OpenGL::GL)
set(SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../Source")
set(NETWORKING_SOURCES "${SOURCE}/EpicExchange.cpp" "${SOURCE}/BackendClient.cpp" "${SOURCE}/LobbyRuntime.cpp" "${SOURCE}/SessionHistory.cpp"
    "${SOURCE}/TransportRouter.cpp" "${SOURCE}/EcomOwnership.cpp" "${SOURCE}/EosTransport.cpp"
    # Replication is carrier-agnostic, so it links the same whether Photon is real or stubbed.
    "${SOURCE}/ReplicationLink.cpp" "${SOURCE}/PhotonReplicationLink.cpp" "${SOURCE}/ReplicationSequence.cpp")
if(PHOTON_LINKED)
    list(APPEND NETWORKING_SOURCES "${SOURCE}/PhotonTransport.cpp")
else()
    list(APPEND NETWORKING_SOURCES "${SOURCE}/PhotonLinkStub.cpp")
endif()
add_executable(NetworkingLogin WIN32 "${SOURCE}/WindowHost.cpp" ${NETWORKING_SOURCES})
target_include_directories(NetworkingLogin PRIVATE "${EOS_SDK_ROOT}/Include" "${PHOTON_INCLUDE}")
target_link_libraries(NetworkingLogin PRIVATE LoginImGui)
# winhttp stays unconditional: the backend premium client needs it in every build.
target_link_libraries(NetworkingLogin PRIVATE winhttp)
if(PHOTON_LINKED)
    target_link_libraries(NetworkingLogin PRIVATE ${PHOTON_LIBS})
    # Photon archives need these OS libs named explicitly (no #pragma comment in headers).
    target_link_libraries(NetworkingLogin PRIVATE winhttp Bcrypt ws2_32 winmm)
endif()
# Compile-now guarantee: the real transport always type-checks against the pinned headers.
add_library(PhotonCompileCheck OBJECT "${SOURCE}/PhotonTransport.cpp")
target_include_directories(PhotonCompileCheck PRIVATE "${PHOTON_INCLUDE}")
if(WIN32)
    target_compile_options(PhotonCompileCheck PRIVATE /W4 /utf-8)
    target_compile_definitions(PhotonCompileCheck PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()
add_dependencies(NetworkingLogin PhotonCompileCheck)
if(WIN32)
    target_link_libraries(NetworkingLogin PRIVATE "${EOS_SDK_ROOT}/Lib/EOSSDK-Win64-Shipping.lib" comdlg32 advapi32)
    target_compile_options(NetworkingLogin PRIVATE /W4 /utf-8)
    target_compile_definitions(NetworkingLogin PRIVATE _CRT_SECURE_NO_WARNINGS)
    set(EOS_RUNTIME "${EOS_SDK_ROOT}/Bin/EOSSDK-Win64-Shipping.dll")
else()
    target_link_libraries(NetworkingLogin PRIVATE "${EOS_SDK_ROOT}/Bin/libEOSSDK-Linux-Shipping.so")
    set_target_properties(NetworkingLogin PROPERTIES BUILD_RPATH "$ORIGIN")
    set(EOS_RUNTIME "${EOS_SDK_ROOT}/Bin/libEOSSDK-Linux-Shipping.so")
endif()
set_target_properties(NetworkingLogin PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../Output"
    RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_CURRENT_SOURCE_DIR}/../Output")
add_custom_command(TARGET NetworkingLogin POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${EOS_RUNTIME}" "$<TARGET_FILE_DIR:NetworkingLogin>")

if(WIN32 AND EXISTS "${EOS_SDK_ROOT}/Bin/xaudio2_9redist.dll")
    add_custom_command(TARGET NetworkingLogin POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
        "${EOS_SDK_ROOT}/Bin/xaudio2_9redist.dll" "$<TARGET_FILE_DIR:NetworkingLogin>")
endif()
