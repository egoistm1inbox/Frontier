cmake_minimum_required(VERSION 3.20)
project(LightCompute LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(Root "../.." ABSOLUTE)
find_path(VulkanHeaders vulkan/vulkan.h
    HINTS "${Root}/ExternalPackages/VulkanCompute/Vulkan-Headers-1.4.321/include" "$ENV{VULKAN_SDK}/include"
    REQUIRED)
set(Sources)
foreach(Name ClusterNesting VisibilityCull DiscrepancySequence ReservoirSampling VarianceFiltering
             DistanceField SurfaceCards ReferenceRenderer)
    list(APPEND Sources "${Root}/Frontier/Engine/Rendering/${Name}.cpp")
endforeach()
foreach(Name LogRingStore LogCategories LogStream)
    list(APPEND Sources "${Root}/Frontier/Engine/Diagnostics/${Name}.cpp")
endforeach()
set(FRONTIER_RENDERING_VULKAN_HEADERS "${VulkanHeaders}")
add_subdirectory("${Root}/Frontier/Engine/Rendering" "${CMAKE_CURRENT_BINARY_DIR}/replay-engine")
add_executable(LightComputeValidation Code/LightComputeValidation.cpp ${Sources})
add_executable(PathReplayDomainValidation Code/PathReplayDomainValidation.cpp)
add_executable(PathReplayVulkanValidation Code/PathReplayVulkanValidation.cpp)
target_link_libraries(PathReplayVulkanValidation PRIVATE FrontierPathReplayVulkan)
target_link_libraries(LightComputeValidation PRIVATE FrontierPathReplaySubmission FrontierRadiantFacetDistribution
    FrontierMediumRadiancePayload)
target_link_libraries(PathReplayDomainValidation PRIVATE FrontierPathReplayDomain)
add_executable(PathReplayArchiveValidation Code/PathReplayArchiveValidation.cpp)
add_executable(ReplaySeedCorrelationValidation Code/ReplaySeedCorrelationValidation.cpp)
set_target_properties(ReplaySeedCorrelationValidation PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON)
add_executable(ReplayVarianceFilterValidation Code/ReplayVarianceFilterValidation.cpp
    "${Root}/Frontier/Engine/Rendering/VarianceFiltering.cpp")
target_include_directories(ReplayVarianceFilterValidation PRIVATE "${Root}/Frontier")
add_executable(ReplayRayAccelerationValidation Code/ReplayRayAccelerationValidation.cpp
    "${Root}/Frontier/Engine/Rendering/ClusterNesting.cpp")
target_link_libraries(ReplayRayAccelerationValidation PRIVATE FrontierPathReplayDomain FrontierPathReplayVulkan)
target_include_directories(ReplayRayAccelerationValidation PRIVATE "${Root}/Frontier" "${VulkanHeaders}")
add_executable(ReplayPbrReflectionValidation Code/ReplayPbrReflectionValidation.cpp
    "${Root}/Frontier/Engine/Rendering/ClusterNesting.cpp")
target_link_libraries(ReplayPbrReflectionValidation PRIVATE FrontierPathReplayDomain FrontierPathReplayVulkan)
target_include_directories(ReplayPbrReflectionValidation PRIVATE "${Root}/Frontier" "${VulkanHeaders}")
add_executable(ReplaySubmissionSequenceValidation Code/ReplaySubmissionSequenceValidation.cpp)
target_link_libraries(ReplaySubmissionSequenceValidation PRIVATE FrontierPathReplaySubmission)
target_include_directories(ReplaySubmissionSequenceValidation PRIVATE "${Root}/Frontier" "${VulkanHeaders}")
add_executable(ProductionReSTIRContractValidation Code/ProductionReSTIRContractValidation.cpp)
target_link_libraries(ProductionReSTIRContractValidation PRIVATE FrontierPathReplaySubmission)
target_include_directories(ProductionReSTIRContractValidation PRIVATE "${Root}/Frontier" "${VulkanHeaders}")
add_executable(PhotometricClusterValidation Code/PhotometricClusterValidation.cpp)
target_link_libraries(PhotometricClusterValidation PRIVATE FrontierPathReplaySubmission FrontierAtmosphericOptics
    FrontierPhotometricLights)
target_include_directories(PhotometricClusterValidation PRIVATE "${Root}/Frontier" "${VulkanHeaders}")
add_executable(RadiantFacetValidation Code/RadiantFacetValidation.cpp)
add_executable(MediumRadianceValidation Code/MediumRadianceValidation.cpp)
target_link_libraries(RadiantFacetValidation PRIVATE FrontierRadiantFacetDistribution)
target_link_libraries(MediumRadianceValidation PRIVATE FrontierMediumRadiancePayload)
set_target_properties(PathReplayArchiveValidation RadiantFacetValidation MediumRadianceValidation
    PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON)
target_include_directories(PathReplayDomainValidation PRIVATE "${Root}/Frontier")
target_include_directories(PathReplayArchiveValidation PRIVATE "${Root}/Frontier")
enable_testing()
add_test(NAME PathReplayDomainValidation COMMAND PathReplayDomainValidation)
add_test(NAME PathReplayVulkanValidation COMMAND PathReplayVulkanValidation)
add_test(NAME RadiantFacetValidation COMMAND RadiantFacetValidation)
add_test(NAME MediumRadianceValidation COMMAND MediumRadianceValidation)
add_test(NAME ReplaySeedCorrelationValidation COMMAND ReplaySeedCorrelationValidation)
add_test(NAME ReplayVarianceFilterValidation COMMAND ReplayVarianceFilterValidation)
add_test(NAME ReplayRayAccelerationValidation COMMAND ReplayRayAccelerationValidation)
add_test(NAME ReplayPbrReflectionValidation COMMAND ReplayPbrReflectionValidation)
add_test(NAME ReplaySubmissionSequenceValidation COMMAND ReplaySubmissionSequenceValidation)
add_test(NAME ProductionReSTIRContractValidation COMMAND ProductionReSTIRContractValidation)
add_test(NAME PhotometricClusterValidation COMMAND PhotometricClusterValidation)
add_test(NAME PathReplayArchiveValidation COMMAND PathReplayArchiveValidation
         "${CMAKE_CURRENT_BINARY_DIR}/ReplayArchiveFixture.bin")
# 📝 A generic build can omit the optional generated fixture; preparing it later must force recompilation.
set(MaterialOffsets "${Root}/_ClaudeScratch/build/light-compute/MaterialSceneOffsets.h")
if(EXISTS "${MaterialOffsets}")
    set_source_files_properties(Code/LightComputeValidation.cpp PROPERTIES OBJECT_DEPENDS "${MaterialOffsets}")
endif()
target_include_directories(LightComputeValidation PRIVATE "${Root}/Frontier" "${Root}" "${VulkanHeaders}")
find_package(Threads REQUIRED)
target_link_libraries(LightComputeValidation PRIVATE Threads::Threads ${CMAKE_DL_LIBS})

    target_compile_options(ProductionReSTIRContractValidation PRIVATE /utf-8 /W4)
    target_compile_options(PhotometricClusterValidation PRIVATE /utf-8 /W4)

    target_compile_options(LightComputeValidation PRIVATE -Wall -Wextra -Werror)
    target_compile_options(PathReplayDomainValidation PRIVATE -Wall -Wextra -Werror)

    target_compile_options(PathReplayArchiveValidation PRIVATE -Wall -Wextra -Werror)
