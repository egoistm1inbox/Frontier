cmake_minimum_required(VERSION 3.20)
project(TyreContinuumValidation LANGUAGES CXX)
option(FRONTIER_TYRE_SANITIZERS "Enable address and undefined-behavior sanitizers for validation" OFF)

get_filename_component(Root "../.." ABSOLUTE)
option(FRONTIER_TYRE_DEVICE_VALIDATION "Build the direct Vulkan tyre projection fixture" OFF)
if(FRONTIER_TYRE_DEVICE_VALIDATION)
    find_path(FRONTIER_TYRE_VULKAN_HEADERS vulkan/vulkan.h
        HINTS "${Root}/ExternalPackages/VulkanCompute/Vulkan-Headers-1.4.321/include" "$ENV{VULKAN_SDK}/include" REQUIRED)
    find_library(TyreVulkanLoader NAMES vulkan libvulkan.so.1
        HINTS "${Root}/ExternalPackages/VulkanCompute/SwiftShader" "$ENV{VULKAN_SDK}/lib" REQUIRED)
endif()
add_subdirectory("${Root}/Frontier/Engine/Physics" "${CMAKE_CURRENT_BINARY_DIR}/tyre-engine")
if(FRONTIER_TYRE_DEVICE_VALIDATION)
    add_executable(TyreProjectionVulkanValidation Code/TyreProjectionVulkanValidation.cpp)
    add_executable(TyreSolidVulkanValidation Code/TyreSolidVulkanValidation.cpp)
    foreach(Target TyreProjectionVulkanValidation TyreSolidVulkanValidation)
        target_link_libraries(${Target} PRIVATE FrontierTyreContinuum FrontierTyreProjectionVulkan "${TyreVulkanLoader}")
        if(MSVC)
            target_compile_options(${Target} PRIVATE /utf-8 /W4)
        else()
            target_compile_options(${Target} PRIVATE -Wall -Wextra -Werror -Wno-missing-field-initializers)
        endif()
    endforeach()
endif()
add_executable(TyreContinuumValidation Code/TyreContinuumValidation.cpp)
add_executable(TyreSurfaceProjectionProbe Code/TyreSurfaceProjectionProbe.cpp)
target_link_libraries(TyreContinuumValidation PRIVATE FrontierTyreContinuum)
target_link_libraries(TyreSurfaceProjectionProbe PRIVATE FrontierTyreContinuum)
set_target_properties(TyreContinuumValidation TyreSurfaceProjectionProbe
    PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON)
if(MSVC)
    target_compile_options(TyreContinuumValidation PRIVATE /utf-8 /W4)
    target_compile_options(TyreSurfaceProjectionProbe PRIVATE /utf-8 /W4)
else()
    target_compile_options(TyreContinuumValidation PRIVATE -Wall -Wextra -Werror)
    target_compile_options(TyreSurfaceProjectionProbe PRIVATE -Wall -Wextra -Werror)
endif()
if(FRONTIER_TYRE_SANITIZERS)
    if(MSVC)
        message(FATAL_ERROR "This validation configures GCC/Clang ASan and UBSan")
    endif()
    foreach(Target FrontierTyreContinuum TyreContinuumValidation TyreSurfaceProjectionProbe)
        target_compile_options(${Target} PRIVATE -O1 -g -fno-omit-frame-pointer -fsanitize=address,undefined)
    endforeach()
    foreach(Target TyreContinuumValidation TyreSurfaceProjectionProbe)
        target_link_options(${Target} PRIVATE -fsanitize=address,undefined)
    endforeach()
    if(FRONTIER_TYRE_DEVICE_VALIDATION)
        foreach(Target FrontierTyreProjectionVulkan TyreProjectionVulkanValidation TyreSolidVulkanValidation)
            target_compile_options(${Target} PRIVATE -O1 -g -fno-omit-frame-pointer -fsanitize=address,undefined)
        endforeach()
        target_link_options(TyreProjectionVulkanValidation PRIVATE -fsanitize=address,undefined)
        target_link_options(TyreSolidVulkanValidation PRIVATE -fsanitize=address,undefined)
    endif()
endif()
find_program(NodeExecutable node REQUIRED)
enable_testing()
add_test(NAME TyreContinuumValidation COMMAND "${CMAKE_COMMAND}"
    "-DNODE=${NodeExecutable}" "-DROOT=${Root}" "-DPROBE=$<TARGET_FILE:TyreContinuumValidation>"
    "-DOBSERVATIONS=${CMAKE_CURRENT_BINARY_DIR}/continuum-reference.txt"
    -P "${CMAKE_CURRENT_SOURCE_DIR}/Code/RunTyreContinuum.cmake")
