cmake_minimum_required(VERSION 3.21)
project(IconRepair LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
get_filename_component(FRONTIER_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
option(ICON_SANITIZERS "Instrument proofs and ThorVG with ASan/UBSan (GCC/Clang)" OFF)
if(ICON_SANITIZERS)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
endif()
include(${FRONTIER_ROOT}/Tools/Build/ThorVGSoftware.cmake)
add_executable(IconProof ${FRONTIER_ROOT}/Engine/DisplayPresentation/IconArt.cpp ${FRONTIER_ROOT}/Exhibits/Workbench/IconArt/IconArtProof.cpp)
target_include_directories(IconProof PRIVATE ${FRONTIER_ROOT}/Engine/DisplayPresentation ${FRONTIER_ROOT}/ExternalPackages/stb)
target_link_libraries(IconProof PRIVATE thorvg_static)
add_executable(AtlasProof ${FRONTIER_ROOT}/Engine/DisplayPresentation/IconArt.cpp ${FRONTIER_ROOT}/Engine/DisplayPresentation/IconPresentation.cpp ${FRONTIER_ROOT}/Exhibits/Workbench/IconArt/PresentationProof.cpp ${FRONTIER_ROOT}/ExternalPackages/imgui/imgui.cpp ${FRONTIER_ROOT}/ExternalPackages/imgui/imgui_draw.cpp ${FRONTIER_ROOT}/ExternalPackages/imgui/imgui_tables.cpp ${FRONTIER_ROOT}/ExternalPackages/imgui/imgui_widgets.cpp)
target_include_directories(AtlasProof PRIVATE ${FRONTIER_ROOT}/Engine/DisplayPresentation ${FRONTIER_ROOT}/ExternalPackages/imgui)
target_link_libraries(AtlasProof PRIVATE thorvg_static)

# Bundle only SVG assets. Tests execute outside the repo working directory.
set(ICON_ROOT "${FRONTIER_ROOT}/EngineContent/Icons")
file(GLOB ICON_FILES CONFIGURE_DEPENDS "${ICON_ROOT}/*.svg" "${ICON_ROOT}/ThorVG/*.svg")
foreach(ICON IN LISTS ICON_FILES)
    file(RELATIVE_PATH RELATIVE_ICON "${ICON_ROOT}" "${ICON}")
    configure_file("${ICON}" "${CMAKE_CURRENT_BINARY_DIR}/EngineContent/Icons/${RELATIVE_ICON}" COPYONLY)
endforeach()
configure_file("${ICON_ROOT}/ThorVG/manifest.json" "${CMAKE_CURRENT_BINARY_DIR}/EngineContent/Icons/ThorVG/manifest.json" COPYONLY)
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/unrelated-cwd")
enable_testing()
add_test(NAME IconArt COMMAND IconProof "${ICON_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/evidence" "${CMAKE_CURRENT_BINARY_DIR}/fixtures")
add_test(NAME IconAtlas COMMAND AtlasProof)
set_tests_properties(IconArt IconAtlas PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/unrelated-cwd")
